## Usage

```tsx
import { EvilBarChart } from "@/components/evilcharts/charts/bar-chart";
```

```tsx
<EvilBarChart  
  xDataKey="month"
  barVariant="default"
  stackType="default"
  data={data} 
  chartConfig={chartConfig}
/>
```
