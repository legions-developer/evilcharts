## Usage

```tsx
import { EvilComposedChart } from "@/components/evilcharts/charts/composed-chart";
```

```tsx
const barConfig = {
  revenue: {
    label: "Revenue",
    colors: { light: ["#3b82f6"], dark: ["#6A5ACD"] },
  },
} satisfies ChartConfig;

const lineConfig = {
  profit: {
    label: "Profit",
    colors: { light: ["#10b981"], dark: ["#34d399"] },
  },
} satisfies ChartConfig;

<EvilComposedChart  
  xDataKey="month"
  data={data} 
  barConfig={barConfig}
  lineConfig={lineConfig}
/>
```
