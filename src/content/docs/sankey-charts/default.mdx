## Usage

```tsx
import { EvilSankeyChart, type SankeyData } from "@/components/evilcharts/charts/sankey-chart";
```

```tsx
const data: SankeyData = {
  nodes: [
    { name: "Visit" },
    { name: "Direct-Favourite" },
    { name: "Page-Click" },
    { name: "Detail-Favourite" },
    { name: "Lost" },
  ],
  links: [
    { source: 0, target: 1, value: 3728 },
    { source: 0, target: 2, value: 354170 },
    { source: 2, target: 3, value: 62429 },
    { source: 2, target: 4, value: 291741 },
  ],
};

const chartConfig = {
  Visit: {
    label: "Visit",
    colors: { light: ["#3b82f6"], dark: ["#60a5fa"] },
  },
  "Page-Click": {
    label: "Page Click",
    colors: { light: ["#f59e0b"], dark: ["#fbbf24"] },
  },
  // ... more node configs
} satisfies ChartConfig;

<EvilSankeyChart
  data={data}
  chartConfig={chartConfig}
  isClickable
/>
```

### Link Variants

The sankey chart supports different link coloring strategies through the `linkVariant` prop.

#### Solid Links

<ComponentPreview className="mb-0" title="linkVariant='solid'" name="ex-solid-link-variant-sankey-chart"  />
<Alert> 
  <AlertContent>
    Set `linkVariant` to `"solid"` to use a single color for all links. This creates a clean, minimal look.
  </AlertContent>
</Alert>

#### Source-colored Links

<ComponentPreview className="mb-0" title="linkVariant='source'" name="ex-source-link-variant-sankey-chart"  />
<Alert> 
  <AlertContent>
    Set `linkVariant` to `"source"` to color links based on their source node. This helps trace where flows originate.
  </AlertContent>
</Alert>
