## Usage

```tsx
import { EvilAreaChart } from "@/components/evilcharts/charts/area-chart";
```

```tsx
<EvilAreaChart  
  xDataKey=""
  type="stacked"
  strokeVariant="dashed"
  areaVariant="gradient"
  activeDotVariant="default"
  data={data}
  chartConfig={chartConfig}
/>
```
