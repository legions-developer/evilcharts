## Usage

```tsx
import { EvilLineChart } from "@/components/evilcharts/charts/line-chart";
```

```tsx
<EvilLineChart  
  xDataKey=""
  curveType="monotone"
  strokeVariant="solid" 
  activeDotVariant="default"
  data={data} 
  chartConfig={chartConfig}
/>
```
